private function parseUseStatements(string $src): array
{
    $map = [];
    preg_match_all('/^use\s+([\w\\\\]+)(?:\s+as\s+(\w+))?\s*;/m', $src, $matches, PREG_SET_ORDER);
    foreach ($matches as $row) {
        $fqcn  = $row[1];
        $alias = $row[2] ?? '';
        $short = $alias !== '' ? $alias : (string) substr(strrchr($fqcn, '\\') ?: '\\' . $fqcn, 1);
        $map[$short] = $fqcn;
    }
    return $map;
}

private function resolveTargetEntity(string $attrs, array $useMap, string $namespace): string

        if (str_contains($ref, '\\')) { return $ref; }
        return $useMap[$ref] ?? ($namespace . '\\' . $ref);

**4-tier resolution priority pattern to preserve verbatim** (v1 lines 67–122):

```php
public function resolve(string $fqcn): string
{
    if (isset($this->resolveCache[$fqcn])) { return $this->resolveCache[$fqcn]; }

    // 1. Overrides win outright — operator escape hatch.
    if (isset($this->overrides[$fqcn])) {
        return $this->resolveCache[$fqcn] = (string) $this->overrides[$fqcn];
    }

    // 2. Doctrine entity parser — source code is the authoritative table name.
    if ($this->entityParser !== null) {
        $info = $this->entityParser->getByFqcn($fqcn);
        if ($info !== null && $info->tableName !== '') {
            return $this->resolveCache[$fqcn] = $info->tableName;
        }
    }

    // 3. Derive candidate basenames from the FQCN tail (DB scan fallback).
    $parts = explode('\\', $fqcn);
    $suffix = end($parts);
    $snake = strtolower((string) preg_replace('/(?<!^)[A-Z]/', '_$0', $suffix));
    $candidates = [$snake . 's', $snake, $snake . 'es'];

    $map = $this->scanMap();
    foreach ($candidates as $cand) {
        if (isset($map[$cand])) { return $this->resolveCache[$fqcn] = $map[$cand]; }
    }

    // 4. Fallback: suffix-match across all tables (lameco_websitebundle_* etc.).
    foreach ($this->allTables() as $table) {
        foreach ($candidates as $cand) {
            if (str_ends_with($table, '_' . $cand)) {
                return $this->resolveCache[$fqcn] = $table;
            }
        }
    }

    throw new \InvalidArgumentException("Cannot resolve {$fqcn} to a legacy detail table.");
}
```

$computeDepth = function (int $id) use (&$computeDepth, &$byId, &$depth, &$visiting, $report): int {
    if (isset($depth[$id])) { return $depth[$id]; }
    if (isset($visiting[$id])) {
        // cycle — caller chain detected revisit before depth was finalised
        if ($report) { $report->warn(sprintf('Cycle detected in kuma_nodes at id=%d', $id)); }
        return PHP_INT_MAX;
    }
    $visiting[$id] = true;
    $parentId = $byId[$id]['kuma_parent_id'] ?? null;
    if ($parentId === null || (int) $parentId === 0) { $d = 0; }
    elseif (!isset($byId[(int) $parentId])) { /* orphan */ $d = PHP_INT_MAX; }
    else {
        $parentDepth = $computeDepth((int) $parentId);
        $d = $parentDepth === PHP_INT_MAX ? PHP_INT_MAX : 1 + $parentDepth;
        if ($d > self::MAX_DEPTH_GUARD) { $d = PHP_INT_MAX; }
    }
    unset($visiting[$id]);
    $depth[$id] = $d;
    return $d;
};

    if (!is_dir($dir) && !@mkdir($dir, 0775, true) && !is_dir($dir)) { return false; }
    $tmp = $path . '.tmp.' . bin2hex(random_bytes(4));
