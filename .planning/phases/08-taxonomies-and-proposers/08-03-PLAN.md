    ```php
    public function renderTaxonomiesMarkdown(?array $mapping, \DateTimeInterface $now): string
    {
        $out = [];
        $out[] = '# Kunstmaan Taxonomy Candidates (non-Page Doctrine entities, generated ' . $now->format(DATE_ATOM) . ')';
        $out[] = '';
        $out[] = '_Non-Page Doctrine entities discovered by KunstmaanSourceScanner. Categories, tags, and other standalone entities live here._';
        $out[] = '_Each entity may carry Gedmo Translatable properties — listed per FQCN._';
        $out[] = '';

        try {
            $scan = $this->kunstmaanSourceScanner->scan();
        } catch (\Throwable $e) {
            $out[] = sprintf('_Could not load source scanner: %s_', $e->getMessage());
            return implode("\n", $out);
        }

        $entities    = (array) ($scan['entities'] ?? []);
        $pageStruct  = (array) ($scan['pageStructure'] ?? []);  // exclude page entities
        $candidates  = [];
        foreach ($entities as $fqcn => $entityInfo) {
            if (!is_string($fqcn) || isset($pageStruct[$fqcn])) {
                continue;
            }
            $candidates[$fqcn] = $entityInfo;
        }

        if ($candidates === []) {
            $out[] = '_No non-Page entities discovered._';
            return implode("\n", $out);
        }

        ksort($candidates);
        foreach ($candidates as $fqcn => $entityInfo) {
            $shortName  = (string) (substr((string) strrchr('\\' . $fqcn, '\\'), 1) ?: $fqcn);
            $sourceTable = (string) ($entityInfo['tableName'] ?? '');
            $rowCount = '?';
            if ($sourceTable !== '' && preg_match('/^[a-z0-9_]+$/i', $sourceTable)) {
                try {
                    $rowCount = (string) (int) $this->legacyDb->queryScalar('SELECT COUNT(*) FROM `' . $sourceTable . '`');
                } catch (\Throwable) {
                    $rowCount = '?';
                }
            }
            $out[] = sprintf('## %s', $shortName);
            $out[] = '';
            $out[] = sprintf('- FQCN: `%s`', $fqcn);
            $out[] = sprintf('- Source table: `%s`', $sourceTable);
            $out[] = sprintf('- Row count: %s', $rowCount);
            $out[] = '';
            // columns table — reuse existing private
            $columnsBlock = $this->renderTableColumns($sourceTable, (array) ($entityInfo['columns'] ?? []));
            if ($columnsBlock !== '') {
                $out[] = $columnsBlock;
                $out[] = '';
            }
            // Gedmo translatable property list (Plan 02's parser flag)
            $translatable = [];
            foreach ((array) ($entityInfo['columns'] ?? []) as $col) {
                if (is_array($col) && (bool) ($col['isGedmoTranslatable'] ?? false)) {
                    $translatable[] = (string) ($col['name'] ?? $col['propertyName'] ?? '');
                }
            }
            if ($translatable !== []) {
                $out[] = '### Gedmo translatable fields';
                $out[] = '';
                foreach ($translatable as $field) {
                    $out[] = '- `' . $field . '`';
                }
                $out[] = '';
            }
        }

        $rendered = implode("\n", $out);
        if (strlen($rendered) > 8000) {
            $rendered = substr($rendered, 0, 8000) . "\n_…truncated…_";
        }
        return $rendered;
    }
    ```
