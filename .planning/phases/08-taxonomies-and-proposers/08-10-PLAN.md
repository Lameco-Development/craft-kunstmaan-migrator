    Step B — extend the constructor (or factory) to accept the FK index / relation graph. The simplest shape is a `Map<FQCN, list<related FQCN>>` derived from the page-structure scanner output (each page's allowed page-part classes + each page-part's relation columns).

    Concrete implementation pattern (adapt to actual VO shape):

    ```php
    /**
     * @param list<string> $allowedFqcns      Explicit allow-list from --entities=.
     * @param array<string, list<string>> $relationGraph  Map<FQCN, list<related-FQCN>>.
     */
    public function __construct(
        public readonly array $allowedFqcns = [],
        // ... existing fields (locales, since, etc.)
        private readonly array $relationGraph = [],
    ) {
    }

    private ?array $reachableCache = null;

    /**
     * Returns the set of FQCNs reachable from $allowedFqcns via the relation graph.
     */
    private function reachableFqcns(): array
    {
        if ($this->reachableCache !== null) {
            return $this->reachableCache;
        }
        $reachable = array_flip($this->allowedFqcns);
        $stack = $this->allowedFqcns;
        while ($stack !== []) {
            $cur = array_pop($stack);
            foreach ($this->relationGraph[$cur] ?? [] as $rel) {
                if (!isset($reachable[$rel])) {
                    $reachable[$rel] = true;
                    $stack[] = $rel;
                }
            }
        }
        return $this->reachableCache = $reachable;
    }

    public function allows(string $fqcn): bool
    {
        if ($this->allowedFqcns === []) {
            return true;  // no scoping applied
        }
        return isset($this->reachableFqcns()[$fqcn]);
    }
    ```
