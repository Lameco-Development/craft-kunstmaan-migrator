    ```php
    /**
     * Phase 8 / D-11, D-12, D-14. Proposes Matrix layout slots for each accepted nodeClass
     * when its entry-type's Matrix catalog has a header- or wrap-shaped block.
     *
     * @param array<string, array> $pageStructure
     * @param array<string, list<string>> $matrixCatalog  Map entryTypeHandle => list<blockHandle>.
     * @return list<array<string, mixed>>
     */
    public function proposeLayoutBlocks(
        array $pageStructure,
        array $matrixCatalog,
        string $kbLegacyMd,
        string $kbCraftMd,
        ?callable $onChunk = null,
    ): array {
        if ($pageStructure === [] || $matrixCatalog === []) {
            return [];
        }
        $apiKey = (string) (Plugin::getInstance()->getSettings()->anthropicApiKey ?? '');
        if ($apiKey === '') {
            throw new MappingProposalException(
                'ANTHROPIC_API_KEY is not set. Set it in .env or plugin settings, or re-run with --no-ai.',
            );
        }

        // Heuristic-trigger filter (D-12).
        $triggered = [];
        foreach ($pageStructure as $fqcn => $info) {
            $handle = (string) ($info['targetEntryType'] ?? $info['entryTypeHandle'] ?? '');
            $blockHandles = (array) ($matrixCatalog[$handle] ?? []);
            $hasHeaderShape = false;
            $hasWrapShape   = false;
            foreach ($blockHandles as $bh) {
                if (preg_match('/^(header|hero|banner)/i', (string) $bh)) { $hasHeaderShape = true; }
                if (preg_match('/^(wrap|container|section)/i', (string) $bh)) { $hasWrapShape = true; }
            }
            if (!$hasHeaderShape && !$hasWrapShape) { continue; }
            $triggered[] = ['fqcn' => $fqcn, 'targetEntryType' => $handle, 'matrixBlocks' => $blockHandles] + (array) $info;
        }
        if ($triggered === []) {
            return [];
        }

        $client  = $this->httpClient ?? \Craft::createGuzzleClient();
        $model   = (string) (Plugin::getInstance()->getSettings()->anthropicModel ?? 'claude-sonnet-4-5');
        $timeout = (int) (Plugin::getInstance()->getSettings()->anthropicTimeout ?? 120);

        $chunks = array_chunk($triggered, 8);
        $chunkTotal = count($chunks);
        $all = [];
        $i = 0;
        foreach ($chunks as $chunk) {
            $i++;
            $startedAt = microtime(true);
            $proposals = $this->proposeLayoutBlocksChunk(
                $chunk, $kbLegacyMd, $kbCraftMd,
                $client, $apiKey, $model, $timeout,
            );
            $all = array_merge($all, $proposals);
            if ($onChunk !== null) {
                $onChunk($i, $chunkTotal, count($chunk), count($proposals), microtime(true) - $startedAt);
            }
        }
        return $all;
    }
    ```
