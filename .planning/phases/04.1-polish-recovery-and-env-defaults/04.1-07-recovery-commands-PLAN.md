    ```php
    public function isTerminal(string $source, string $key, ?int $siteId): bool
    {
        $row = $this->get($source, $key, $siteId);
        if (!$row) {
    ```

    **3a. Add `actionSyncAssets()` to `MigrateController`** (place after `actionTruncate` or another sync sibling — preserve actionIndex / actionExtract / etc. order):

    ```php
    /**
     * Phase 4.1 / REC-01 — re-ingest every kuma_media row prior atomic runs
     * referenced but skipped (filesystem_404 / mime_mismatch / too_large / etc.).
     * Idempotent: re-running with everything healed exits cleanly.
     * Permanently-failed rows (meta.terminalState='permanently_failed') reported
     * but NOT retried. Respects MigrationFilters (--entity / --locale / --since).
     */
    public function actionSyncAssets(): int
    {
        if (($gate = $this->enforceNeverProduction()) !== null) {
            return $gate;
        }
        $this->openLogFile($this->defaultLogPath());
        $this->logLine('actionSyncAssets started; verbosity=' . $this->verbosityLevel(), 1);

        $plugin     = Plugin::getInstance();
        $filters    = $plugin->filterFactory->fromCli(
            $this->entities,
            $this->locales,
            $this->since,
            $this->noSeo,
            $this->noRetour,
        );
        $storageDir = Craft::$app->path->getStoragePath() . '/migration';
        $report     = new MigrationReport();
        $tStart     = microtime(true);

        $this->stdout("Sync-assets: re-ingesting skipped media rows\n", Console::FG_CYAN);

        $candidates = $this->collectAssetCandidates($filters);
        $this->stdout(sprintf("  Found %d candidates\n", count($candidates)), Console::FG_GREEN);

        foreach ($candidates as $i => $candidate) {
            $this->logLine(sprintf('[%d/%d] media=%s', $i + 1, count($candidates), $candidate['sourceKey']), 1);
            try {
                $asset = $plugin->assetMigrationService->resolveFromLegacyId(
                    (int) $candidate['legacyId'],
                    $candidate['siteId'] ?? null,
                );
                if ($asset === null) {
                    // resolveFromLegacyId returned null without throwing — treat as retryable.
                    $report->incr('failed');
                    $this->stdout(sprintf("  FAIL %s — resolve returned null (will retry on next run)\n", $candidate['sourceKey']), Console::FG_YELLOW);
                } else {
                    $report->incr('healed');
                    $this->stdout(sprintf("  OK   %s → asset:%d\n", $candidate['sourceKey'], $asset->id), Console::FG_GREEN);
                }
            } catch (\Throwable $e) {
                // Phase 4.1 / D-37 — inline failure classification, mirroring
                // AssetMigrationService::classifyAssetFailureReason heuristics
                // (lines 688-707). We deliberately do NOT add a public
                // lastFailureReason() to AssetMigrationService — keeping the
                // classifier private preserves the existing surface.
                $reason = self::classifyResolveFailureMessage($e->getMessage());
                if (in_array($reason, ['filesystem_404', 'too_large'], true)) {
                    $plugin->migrationStateService->markTerminal(
                        'media',
                        $candidate['sourceKey'],
                        $candidate['siteId'] ?? null,
                        $reason,
                    );
                    $report->incr('terminal');
                    $this->stdout(sprintf("  TERM %s — %s\n", $candidate['sourceKey'], $reason), Console::FG_RED);
                } else {
                    $report->incr('failed');
                    $this->stderr(sprintf("  FAIL %s — %s (reason=%s, will retry on next run)\n", $candidate['sourceKey'], $e->getMessage(), $reason), Console::FG_RED);
                }
            }
        }

        // (See classifyResolveFailureMessage helper — defined below alongside collectAssetCandidates.)

        // D-39: emit SYNC-ASSETS-<timestamp>.md
        $reportPath = $storageDir . '/SYNC-ASSETS-' . gmdate('Y-m-d--H-i-s') . '.md';
        $rendered   = self::renderSyncReport('Sync Assets', $report, $filters, $tStart, $candidates);
        if (!$plugin->mappingFile->writeAtomic($reportPath, $rendered)) {
            $this->stderr("  FAIL could not write {$reportPath}\n", Console::FG_RED);
            return ExitCode::UNSPECIFIED_ERROR;
        }
        $this->stdout("  OK   SYNC-ASSETS report written → {$reportPath}\n", Console::FG_GREEN);

        return $report->incrFor('failed') > 0 ? ExitCode::UNSPECIFIED_ERROR : ExitCode::OK;
    }

    /**
     * @return list<array{legacyId:int, sourceKey:string, siteId:?int, ...}>
     */
    private function collectAssetCandidates(MigrationFilters $filters): array
    {
        $plugin = Plugin::getInstance();
        $rows   = $plugin->migrationStateService->all('media');
        $result = [];
        foreach ($rows as $row) {
            // D-37: terminal rows are never retried.
            if ($plugin->migrationStateService->isTerminal('media', $row['sourceKey'], $row['siteId'] ?? null)) {
                continue;
            }
            // D-36 idempotence: skip rows whose targetId is set (already healed).
            if (!empty($row['targetId'])) {
                continue;
            }
            // Filter scope (D-38).
            if (!self::candidateMatchesFilters($row, $filters)) {
                continue;
            }
            $legacyId = (int) ($row['sourceKey'] ?? 0);
            if ($legacyId <= 0) {
                continue;
            }
            $result[] = [
                'legacyId'  => $legacyId,
                'sourceKey' => (string) $row['sourceKey'],
                'siteId'    => $row['siteId'] ?? null,
                'meta'      => $row['meta'] ?? null,
            ];
        }
        return $result;
    }

    private static function candidateMatchesFilters(array $row, MigrationFilters $filters): bool
    {
        // entities filter — for asset rows, this means scoping to assets referenced
        // by entries of the named entity types. Reuse the ExtractService convention.
        if ($filters->entities !== []) {
            $rowEntity = $row['meta']['ownerEntity'] ?? null;
            if ($rowEntity !== null && !in_array((string) $rowEntity, $filters->entities, true)) {
                return false;
            }
        }
        // since filter
        if ($filters->since !== null) {
            $rowDate = $row['dateUpdated'] ?? null;
            if ($rowDate !== null && strtotime((string) $rowDate) < strtotime($filters->since)) {
                return false;
            }
        }

    /**
     * Phase 4.1 / D-37 — inline failure classification.
     * Mirrors AssetMigrationService::classifyAssetFailureReason (private;
     * lines 688-707) so we never need to expose that classifier publicly.
     * Pure string-pattern function; testable via Reflection.
     */
    private static function classifyResolveFailureMessage(string $msg): string
    {
        if (str_contains($msg, 'No such file')
            || str_contains($msg, 'not found')
            || str_contains($msg, 'Copy failed')
        ) {
            return 'filesystem_404';
        }
        if (str_contains($msg, 'mime')
            || str_contains($msg, 'content_type')
            || str_contains($msg, 'allowedFileExtensions')
        ) {
            return 'mime_mismatch';
        }
        if (str_contains($msg, 'too large') || str_contains($msg, 'PostMaxSize')) {
            return 'too_large';
        }
        return 'deferred_unresolved';
    }

    /**
     * Phase 4.1 / D-39 — render SYNC-*.md body.
     * Format mirrors REPORT.md rehearsal-summary section.
     */
    private static function renderSyncReport(
        string $title,
        MigrationReport $report,
        MigrationFilters $filters,
        float $tStart,
        array $candidates,
    ): string {
        $lines   = [];
        $lines[] = "# {$title}";
        $lines[] = '';
        $lines[] = '## Rehearsal summary';
        $lines[] = '';
        $lines[] = sprintf('- Candidates: %d', count($candidates));
        $lines[] = sprintf('- Healed:     %d', $report->incrFor('healed'));
        $lines[] = sprintf('- Failed:     %d', $report->incrFor('failed'));
        $lines[] = sprintf('- Terminal:   %d', $report->incrFor('terminal'));
        $lines[] = sprintf('- Wall clock: %s', gmdate('H:i:s', (int) (microtime(true) - $tStart)));
        $lines[] = sprintf('- Filters:    entities=%s; locales=%s; since=%s',
            $filters->entities === [] ? 'all' : implode(',', $filters->entities),
            $filters->locales === [] ? 'all' : implode(',', $filters->locales),
            $filters->since ?? 'none',
        );
        $lines[] = '';
        if ($candidates === []) {
            $lines[] = '_No candidates — all prior skipped media has been healed or marked terminal._';
        }
        return implode("\n", $lines) . "\n";
    }
    ```
