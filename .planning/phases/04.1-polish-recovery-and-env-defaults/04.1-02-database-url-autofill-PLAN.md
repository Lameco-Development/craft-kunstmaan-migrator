    Add the new public method `beforeValidate()` (per D-07) — per the PATTERNS.md excerpt:

        /**
         * Phase 4.1 / D-07 — auto-fill blank legacyDb* from .env DATABASE_URL.
         * Operator-supplied Settings always win; only blank slots fill.
         * D-08: parse_url + urldecode per component; D-09: only mysql schemes honored.
         */
        public function beforeValidate(): bool
        {
            // Fast path: if all 4 string fields are non-blank, skip env reader entirely.
            if ($this->legacyDbServer !== null && $this->legacyDbServer !== ''
                && $this->legacyDbDatabase !== null && $this->legacyDbDatabase !== ''
                && $this->legacyDbUser !== null && $this->legacyDbUser !== ''
                && $this->legacyDbPassword !== null && $this->legacyDbPassword !== ''
            ) {
                return parent::beforeValidate();
            }

            $reader = $this->getEnvReader();
            $dsn = $reader->getDatabaseUrl();
            if ($dsn === null || $dsn === '') {
                return parent::beforeValidate();
            }

            // D-09: reader returns null parsed components for non-mysql schemes.
            // If parsedDsn is null, every getDsn* accessor returns null → all ??= no-ops.
            if ($reader->getDsnHost() === null && $reader->getDsnUser() === null && $reader->getDsnDatabase() === null) {
                return parent::beforeValidate();
            }

            // D-07: operator values win. Use ??= (only fill null).
            // Empty-string treated as operator-provided (operator might intentionally blank a field; respect it).
            $this->legacyDbServer   ??= $reader->getDsnHost();
            $this->legacyDbUser     ??= $reader->getDsnUser();
            $this->legacyDbPassword ??= $reader->getDsnPassword();
            $this->legacyDbDatabase ??= $reader->getDsnDatabase();

            // Port special case: default 3306 is the sentinel "operator hasn't customized."
            // Only override default 3306 when DSN provides an explicit port.
            $dsnPort = $reader->getDsnPort();
            if ($dsnPort !== null && $this->legacyDbPort === 3306) {
                $this->legacyDbPort = $dsnPort;
            }

            return parent::beforeValidate();
        }
